## 📊 Модуль 1: KPI Менеджеров

### 1.1 Метрики эффективности

| Метрика | Формула | Вес | Норма |
|---------|---------|-----|-------|
| **Конверсия** | Сделки / Лиды × 100% | 30% | ≥40% |
| **Средний чек** | Выручка / Сделки | 25% | ≥150,000 ₸ |
| **Скорость ответа** | Среднее время первого ответа | 20% | ≤5 мин |
| **Вовлечённость** | Сообщений от менеджера / Всего сообщений | 15% | 40-60% |
| **Качество коммуникации** | AI-оценка тональности, скрипта | 10% | ≥7/10 |

### 1.2 Расчёт итогового скора

```typescript
interface ManagerScore {
  managerId: string;
  managerName: string;

  // Raw metrics
  metrics: {
    totalLeads: number;
    totalDeals: number;
    totalRevenue: number;
    avgResponseTime: number; // minutes
    messageCount: number;
    aiQualityScore: number; // 1-10
  };

  // Calculated scores (0-100)
  scores: {
    conversion: number;      // (deals/leads) normalized
    avgCheck: number;        // (revenue/deals) normalized
    responseSpeed: number;   // inverse of response time
    engagement: number;      // message ratio
    quality: number;         // AI score × 10
  };

  // Final score
  totalScore: number;        // weighted average
  rank: 'S' | 'A' | 'B' | 'C' | 'D';
  trend: 'up' | 'down' | 'stable';
}

// Score calculation
function calculateManagerScore(metrics: ManagerMetrics): number {
  const weights = {
    conversion: 0.30,
    avgCheck: 0.25,
    responseSpeed: 0.20,
    engagement: 0.15,
    quality: 0.10
  };

  // Normalize each metric to 0-100
  const scores = {
    conversion: normalize(metrics.conversion, 0, 60, 0, 100),
    avgCheck: normalize(metrics.avgCheck, 50000, 300000, 0, 100),
    responseSpeed: normalize(30 - metrics.avgResponseTime, 0, 30, 0, 100),
    engagement: normalize(metrics.engagement, 20, 60, 0, 100),
    quality: metrics.aiQualityScore * 10
  };

  // Weighted sum
  return Object.entries(weights).reduce((sum, [key, weight]) =>
    sum + scores[key] * weight, 0
  );
}

// Rank assignment
function getRank(score: number): string {
  if (score >= 90) return 'S'; // Супер
  if (score >= 75) return 'A'; // Отлично
  if (score >= 60) return 'B'; // Хорошо
  if (score >= 45) return 'C'; // Удовлетворительно
  return 'D';                   // Требует внимания
}
```

type RecommendationType =
  | 'OPTIMIZE_CREATIVE'      // Оптимизация креативов
  | 'ADJUST_TARGETING'       // Улучшение таргетинга
  | 'REALLOCATE_BUDGET'      // Перераспределение бюджета
  | 'MANAGER_COACHING'       // Обучение менеджера
  | 'SCALE_CAMPAIGN'         // Масштабирование кампании
  | 'PAUSE_CAMPAIGN'         // Остановка кампании
  | 'A_B_TEST'               // Запуск A/B теста
  | 'AUDIENCE_EXPANSION';    // Расширение аудитории

interface AIRecommendation {
  id: string;
  type: RecommendationType;
  priority: 'critical' | 'high' | 'medium' | 'low';

  target: {
    type: 'manager' | 'campaign' | 'creative' | 'placement';
    id: string;
    name: string;
  };

  insight: string;           // Что обнаружено
  recommendation: string;    // Что делать
  expectedImpact: string;    // Ожидаемый результат

  actions: {
    label: string;
    action: () => void;
  }[];

  confidence: number;        // 0-100% уверенность AI
  basedOn: string[];         // На каких данных основано

### 5.3 Интеграция Claude API

```typescript
// AI Analyzer Service
interface ClaudeAnalysisRequest {
  context: {
    managers: ManagerScore[];
    campaigns: CampaignMetrics[];
    creatives: CreativeScore[];
    placements: PlacementScore[];
    historicalData: HistoricalMetrics[];
  };

  questions: [
    'Какие кампании сливают бюджет?',
    'Какие менеджеры нуждаются в обучении?',
    'Какие креативы нужно масштабировать?',
    'Как оптимизировать распределение бюджета?'
  ];
}

interface ClaudeAnalysisResponse {
  recommendations: AIRecommendation[];
  summary: string;
  forecasts: {
    metric: string;
    current: number;
    predicted: number;
    confidence: number;
  }[];
}

async function analyzeWithClaude(data: ClaudeAnalysisRequest): Promise<ClaudeAnalysisResponse> {
  const response = await fetch('/api/ai/analyze', {
    method: 'POST',
    body: JSON.stringify({
      model: 'claude-3-opus',
      system: `Ты — AI аналитик для РОП (руководителя отдела продаж).
        Анализируй данные и давай конкретные рекомендации.
        Фокусируйся на: ROI, конверсии, эффективности команды.
        Отвечай на русском языке.`,
      data
    })
  });

  return response.json();
}
```
